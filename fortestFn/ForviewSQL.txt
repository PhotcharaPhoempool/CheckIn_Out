SELECT 
    a.id,
    e.employee_code,
    CONCAT(e.title, ' ', e.first_name, ' ', e.last_name) AS full_name,
    a.check_time,
    a.status,
    a.camera_name
FROM attendance_logs a
JOIN employees e ON a.employee_code = e.employee_code
ORDER BY a.check_time DESC;

#Reset data
TRUNCATE TABLE attendance_logs RESTART IDENTITY;
